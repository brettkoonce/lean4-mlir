% This file makes a printable version of the blueprint
% It should include all the \usepackage needed for the pdf version.
% The template version assume you want to use a modern TeX compiler
% such as xeLaTeX or luaLaTeX including support for unicode
% and Latin Modern Math font with standard bugfixes applied.
% It also uses expl3 in order to support macros related to the dependency graph.
% It also includes standard AMS packages (and their improved version
% mathtools) as well as support for links with a sober decoration
% (no ugly rectangles around links).
% It is otherwise a very minimal preamble (you should probably at least
% add cleveref and tikz-cd).

\documentclass[a4paper]{report}

\usepackage{fontspec}
% Latin Modern Mono has no U+2218, box-drawing, lowercase Greek or superscripts,
% so verbatim transcripts lost characters silently (pretty∘emit rendered as
% pretty?emit). DejaVu Sans Mono covers all of them. Scaled to match the
% surrounding Latin Modern x-height, which DejaVu overshoots at nominal size.
\setmonofont{DejaVu Sans Mono}[Scale=MatchLowercase]
\usepackage{geometry}
\geometry{margin=0.9in}

% Shrink verbatim font so long NetSpec comments don't wrap at default width.
\makeatletter
\renewcommand{\verbatim@font}{\ttfamily\small}
\makeatother

\usepackage{expl3}

\usepackage{graphicx}
\graphicspath{{figures/}}

\usepackage{amssymb, amsthm, mathtools}
\usepackage[unicode,colorlinks=true,linkcolor=blue,urlcolor=magenta, citecolor=blue]{hyperref}

\usepackage[warnings-off={mathtools-colon,mathtools-overbracket}]{unicode-math}

\usepackage{tikz}
\usetikzlibrary{positioning, arrows.meta, calc, fit}
\usepackage{pgfplots}
\pgfplotsset{compat=1.16}

\input{macros/common}
\input{macros/print}

\title{Verified Deep Learning with Lean 4 \\[0.6ex]
  {\large Formal Backpropagation from MLP to Attention, via MLIR}}
\author{Brett Koonce}
\date{Version 0.6.3}

\begin{document}
\maketitle
% Chapters and numbered sections only. The subsection and \paragraph
% headings are chapter-internal signposts; in a 100-section book they
% triple the TOC's length without helping anyone navigate it.
\setcounter{tocdepth}{1}
\tableofcontents
\input{content}
\end{document}
